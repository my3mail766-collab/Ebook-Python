9. OOP (Class & Objek)

class Komponen:
    def __init__(self, nama, harga, stok):
        self.nama = nama
        self.harga = harga
        self.stok = stok

    def jual(self, jumlah):
        if jumlah > self.stok:
            print("Stok kurang")
            return 0
        self.stok -= jumlah
        return self.harga * jumlah

class Laptop(Komponen):          # pewarisan
    def __init__(self, nama, harga, stok, layar):
        super().__init__(nama, harga, stok)
        self.layar = layar

ram = Komponen("RAM 16GB", 650000, 10)
print(ram.jual(2), ram.stok)

    for i in range(n):

# decorator
def catat(fungsi):
    def bungkus(*a, **k):
        print("Memanggil", fungsi.__name__)
        return fungsi(*a, **k)
    return bungkus

@catat
def salam(nama):
    return "Halo " + nama
print(salam("Andi"))

12. Proyek: Kalkulator Rakitan PC

Menggabungkan dict, fungsi, loop, dan file JSON menjadi aplikasi kecil.

import json

KATALOG = {
    "1": ("Ryzen 5 5600", 1850000),
    "2": ("RAM 16GB DDR4", 650000),
    "3": ("SSD 512GB", 700000),
    "4": ("PSU 550W", 550000),
}

def pilih_komponen():
    keranjang = []
    while True:
        for k, (nama, h) in KATALOG.items():
            print(f"{k}. {nama} - Rp{h:,}")
        pilih = input("Pilih nomor (Enter untuk selesai): ")
        if pilih == "":
            return keranjang
        if pilih in KATALOG:
            keranjang.append(KATALOG[pilih])
        else:
            print("Nomor tidak valid")

def hitung(keranjang, jasa=150000):
    return sum(h for _, h in keranjang) + jasa

def simpan(pelanggan, keranjang, total):
    data = {"pelanggan": pelanggan,
            "item": [n for n, _ in keranjang],
            "total": total}
    with open("pesanan.json", "a") as f:
        f.write(json.dumps(data) + "\n")

pelanggan = input("Nama pelanggan: ")
item = pilih_komponen()
total = hitung(item)
print(f"Total (termasuk jasa rakit): Rp{total:,}")
simpan(pelanggan, item, total)
